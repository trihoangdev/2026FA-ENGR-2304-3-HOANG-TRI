#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
#include<string.h>
#include<math.h>

#define NUM_PLANTS	6
#define MAX_RECORD	5000
#define MAX_VALUE   1000  
#define DEFAUT_FILE "data.csv"

typedef struct {
	int year, month, day;
	int units[NUM_PLANTS];
}Record;

typedef struct {
	int    min, max, mode;
	double mean, variance, stddev;
	long   total;
} Stats;

/* -------------- FUNCTIONS ------------------*/
int load_data(const char* filename, Record rec[]);
void calc_stats(const Record rec[], int n, int plant, Stats* s);
void plant_analysis(const Record rec[], int n);
void production_report(const Record rec[], int n);
void search_data(const Record rec[], int n);
void print_date(const Record* r);
void clear_input(void);

int main(void)
{
	static Record rec[MAX_RECORD];
	char filename[256] = DEFAUT_FILE;
	int n = load_data(filename, rec);

	/* if the default file is missing, ask the user for the name */
	while (n <= 0) {
		printf("Cannot open '%s'. Enter file name (or q to quit): ", filename);
		if (scanf("%259s", filename) != 1 || strcmp(filename, "q") == 0)
			return 1;
		n = load_data(filename, rec);
	}
	printf("Loaded %d daily records from '%s'.\n", n, filename);

	int choice;
	do {
		printf("\n==============================\n");
		printf("            MENU\n");
		printf("==============================\n");
		printf(" 1. Plant Data Analyses\n");
		printf(" 2. Production Report\n");
		printf(" 3. Search Production Data\n");
		printf(" 4. Exit\n");
		printf("Select an option (1-4): ");

		if (scanf("%d", &choice) != 1) {
			clear_input();
			choice = 0;
		}
		switch (choice) {
		case 1: plant_analysis(rec, n);     break;
		case 2: production_report(rec, n);  break;
		case 3: search_data(rec, n);        break;
		case 4: printf("Goodbye!\n");       break;
		default: printf("Invalid choice, please enter 1-4.\n");
		}
	} while (choice != 4);
	return 0;
}

/* Read the CSV file. Returns number of records, or -1 if file not found. */
int load_data(const char* filename, Record rec[])
{
	FILE* fp;
	fp = fopen(filename, "r");
	char line[256];
	int n = 0;

	if (fp == NULL) return -1; //file empty
	fgets(line, sizeof line, fp); //Ignore "Year, Month,..."

	while (n < MAX_RECORD && fgets(line, sizeof line, fp) != NULL)
	{
		Record* r = &rec[n];

		//Change ',' -> ' '
		for (char* p = line; *p; p++)
		{
			if (*p == ',')
				*p = ' ';
		}

		int got = sscanf(line, "%d %d %d %d %d %d %d %d %d",
			&r->year, &r->month, &r->day,
			&r->units[0], &r->units[1], &r->units[2],
			&r->units[3], &r->units[4], &r->units[5]);

		//If data of line < 9 -> ignore that line and overwrite on next line
		if (got == 3 + NUM_PLANTS)
		{
			n++;
		}
	}
	fclose(fp);
	return n;

}

/* Discard the rest of the current input line */
void clear_input(void)
{
	int c;
	while ((c = getchar()) != '\n' && c != EOF)
		;
}

/* Compute min, max, mode, mean, variance and std deviation of one plant */
void calc_stats(const Record rec[], int n, int plant, Stats* s)
{
	static int count[MAX_VALUE + 1];
	double sum_sq = 0.0;
	int i;

	memset(count, 0, sizeof count);
	s->min = s->max = rec[0].units[plant];
	s->total = 0;

	for (i = 0; i < n; i++) {
		int v = rec[i].units[plant];
		if (v < s->min) s->min = v;
		if (v > s->max) s->max = v;
		s->total += v;
		if (v >= 0 && v <= MAX_VALUE)
			count[v]++;
	}
	s->mean = (double)s->total / n;

	for (i = 0; i < n; i++) {
		double d = rec[i].units[plant] - s->mean;
		sum_sq += d * d;
	}
	/* sample variance (divide by n-1) */
	s->variance = (n > 1) ? sum_sq / (n - 1) : 0.0;
	s->stddev = sqrt(s->variance);

	s->mode = 0;
	for (i = 1; i <= MAX_VALUE; i++)
		if (count[i] > count[s->mode])
			s->mode = i;
}

void print_date(const Record* r)
{
	printf("%02d/%02d/%04d", r->month, r->day, r->year);
}


/* Menu 1: descriptive statistics table */
void plant_analysis(const Record rec[], int n)
{
	Stats s;
	printf("\n---------------- PLANT DATA ANALYSES (%d days) ----------------\n", n);
	printf("%-6s %5s %5s %5s %9s %10s %9s\n",
		"Plant", "Min", "Max", "Mode", "Mean", "Variance", "Std Dev");
	for (int p = 0; p < NUM_PLANTS; p++) {
		calc_stats(rec, n, p, &s);
		printf("L%-5d %5d %5d %5d %9.2f %10.2f %9.2f\n",
			p + 1, s.min, s.max, s.mode, s.mean, s.variance, s.stddev);
	}
	printf("(Variance / Std Dev are sample statistics, n-1)\n");
}

/* Menu 2: best / worst plant, best single day */
void production_report(const Record rec[], int n)
{
	Stats s;
	long total[NUM_PLANTS];
	int best_plant = 0, worst_plant = 0;
	int best_day_plant = 0, best_day_idx = 0, best_day_val = -1;
	int best_all_idx = 0, best_all_sum = -1;
	int i, p;

	for (p = 0; p < NUM_PLANTS; p++) {
		calc_stats(rec, n, p, &s);
		total[p] = s.total;
		if (total[p] > total[best_plant])  best_plant = p;
		if (total[p] < total[worst_plant]) worst_plant = p;
	}

	for (i = 0; i < n; i++) {
		int sum = 0;
		for (p = 0; p < NUM_PLANTS; p++) {
			sum += rec[i].units[p];
			if (rec[i].units[p] > best_day_val) {
				best_day_val = rec[i].units[p];
				best_day_plant = p;
				best_day_idx = i;
			}
		}
		if (sum > best_all_sum) {
			best_all_sum = sum;
			best_all_idx = i;
		}
	}

	printf("\n==================== PRODUCTION REPORT ====================\n");
	printf("Records analyzed: %d days (", n);
	print_date(&rec[0]); printf(" to "); print_date(&rec[n - 1]); printf(")\n\n");

	printf("Total units produced per plant:\n");
	for (p = 0; p < NUM_PLANTS; p++)
		printf("  L%d: %ld  (avg %.2f/day)\n", p + 1, total[p], (double)total[p] / n);

	printf("\nBest performing plant (overall):  L%d  - %ld units, avg %.2f/day\n",
		best_plant + 1, total[best_plant], (double)total[best_plant] / n);
	printf("Worst performing plant (overall): L%d  - %ld units, avg %.2f/day\n",
		worst_plant + 1, total[worst_plant], (double)total[worst_plant] / n);

	printf("Best single production day for a single plant: L%d produced %d units on ",
		best_day_plant + 1, best_day_val);
	print_date(&rec[best_day_idx]); printf("\n");

	printf("Best single production day for all facilities: ");
	print_date(&rec[best_all_idx]);
	printf(" - %d units combined\n", best_all_sum);
	printf("   (");
	for (p = 0; p < NUM_PLANTS; p++)
		printf("L%d=%d%s", p + 1, rec[best_all_idx].units[p], p < NUM_PLANTS - 1 ? ", " : "");
	printf(")\n");
}

/* Menu 3: search by date (single plant or all plants) */
void search_data(const Record rec[], int n)
{
	int choice, plant = 0, y, m, d, i, p, found = 0;

	printf("\n--- SEARCH PRODUCTION DATA ---\n");
	printf(" 1. Single plant for a single date\n");
	printf(" 2. All plants for a single date\n");
	printf("Select (1-2): ");
	if (scanf("%d", &choice) != 1 || (choice != 1 && choice != 2)) {
		clear_input();
		printf("Invalid selection.\n");
		return;
	}
	if (choice == 1) {
		printf("Plant number (1-%d): ", NUM_PLANTS);
		if (scanf("%d", &plant) != 1 || plant < 1 || plant > NUM_PLANTS) {
			clear_input();
			printf("Invalid plant number.\n");
			return;
		}
	}
	printf("Enter date as Year Month Day (e.g. 1992 11 14): ");
	if (scanf("%d %d %d", &y, &m, &d) != 3) {
		clear_input();
		printf("Invalid date.\n");
		return;
	}

	for (i = 0; i < n && !found; i++) {
		if (rec[i].year == y && rec[i].month == m && rec[i].day == d) {
			found = 1;
			printf("\nProduction on "); print_date(&rec[i]); printf(":\n");
			if (choice == 1)
				printf("  L%d: %d units\n", plant, rec[i].units[plant - 1]);
			else
				for (p = 0; p < NUM_PLANTS; p++)
					printf("  L%d: %d units\n", p + 1, rec[i].units[p]);
		}
	}
	if (!found)
		printf("No data recorded for %02d/%02d/%04d.\n", m, d, y);
}
